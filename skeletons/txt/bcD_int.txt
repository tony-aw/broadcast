inline bool inline_dims_is_vector(SEXP x) {
  
  if(x == R_NilValue) {
    return true;
  }
  
  const int n = Rf_xlength(x);
  const int *px = INTEGER_RO(x);
  int count_big = 0;
  
  for(int i = 0; i < n; ++i) {
    count_big += px[i] > 1;
  }
  
  
  return (count_big <= 1);
}



inline std::array<long long, 16> inline_make_strides(
  SEXP target_dim
) {
  
  // get properties:
  int n = Rf_length(target_dim);
  std::array<long long, 16> out{1};
  long long *pout = out.data();
  const int *ptarget = INTEGER_RO(target_dim);
  
  
  // make dcp:
  double temp_prod = ptarget[0];
  pout[0] = 1;
  pout[1] = ptarget[0];
  if(n > 2) {
    for(int i = 2; i < n; ++i) {
      temp_prod = temp_prod * ptarget[i-1];
      pout[i] = temp_prod;
    }
  }

  // mulitply with by (i.e. make it zero when appropriate):
  for(int i = 0; i < n; ++i) {
    if(ptarget[i] <= 1) {
      pout[i] = 0;
    }
  }
  
  return out;
  
}




inline int inline_determine_loopdepth(
  SEXP out_dim
) {
  int n = Rf_length(out_dim);
  const int *pod = INTEGER_RO(out_dim);
  int rm = 0;
  for(int i = (n-1); i >= 0; --i) {
    if(pod[i] == 1) {
      rm++;
    }
    else {
      break;
    }
  }
  int loopdepth = n - rm;
  if(loopdepth == 0) {
   loopdepth = 1;
  }
  return loopdepth;
  
}

// ********************************************************************************
// MACROs for broadcasted element-wise binary operations
// 
// The following MACROs define the loops used for broadcasted element-wise binary operations.
// 
// In the context of a broadcasted operation involving exactly 2 arrays,
// 'broadcast' uses different techniques for looping through the elements for broadcasting.
// The techniques are the following, ordered from high to low priority:
//  1) Vector mode:
//   This mode is usedbroadcasting where one or both of the arrays is actually vector or 1d-array.
//   This mode is also used when no broadcasting needs to occur.
//  2) General mode:
//   For regular broadcasting.
// 
// The dimensions of both arrays are first NORMALIZED and SIMPLIFIED virtually
// (see 'rcpp_virt_binary_prep' code),
// before determining which technique to use.
// 
// The "odometer trick" is inspired by various stackoverflow messages and a few blog posts,
// though none of them gives a proper unified view.
// 
// 
// 
// 
// ********************************************************************************
// 
// 


#define MACRO_DIM_VECTOR(DOCODE, GETX, GETY) do {                               \
  R_xlen_t flatind_x = 0;                                           \
  R_xlen_t flatind_y = 0;                                           \
  int by_x = 0;                                                     \
  int by_y = 0;                                                     \
  if(Rf_xlength(x) == Rf_xlength(y)) {                              \
    if(Rf_xlength(x) == 1) {                                        \
      by_x = 0;                                                     \
      by_y = 0;                                                     \
    }                                                               \
    else {                                                          \
      by_x = 1;                                                     \
      by_y = 1;                                                     \
    }                                                               \
  }                                                                 \
  if(Rf_xlength(x) != Rf_xlength(y)) {                              \
    if(Rf_xlength(x) == 1) {                                        \
      by_x = 0;                                                     \
      by_y = 1;                                                     \
    }                                                               \
    else if(Rf_xlength(y) ==1) {                                    \
      by_x = 1;                                                     \
      by_y = 0;                                                     \
    }                                                               \
  }                                                                 \
  for(R_xlen_t flatind_out = 0; flatind_out < nout; ++flatind_out) {  \
    GETX; \
    GETY; \
    DOCODE;                                                           \
    flatind_x += by_x;                                     \
    flatind_y += by_y;                                     \
  }                                                                   \
                                                                      \
} while(0)



#define MACRO_DIM_ORTHOVECTOR(DOCODE, GETX, GETY) do {      \
  R_xlen_t flatind_out = 0;         \
  const int N1 = INTEGER_RO(out_dim)[0];      \
  const int N2 = INTEGER_RO(out_dim)[1];       \
  bool RxC = INTEGER_RO(x_dim)[0] != 1; \
  if(RxC) { \
    for(int flatind_y = 0; flatind_y < N2; ++flatind_y) {	\
      GETY; \
  	  for(int flatind_x = 0; flatind_x < N1; ++flatind_x) {	\
  	    GETX; \
        DOCODE;                         \
        ++flatind_out;                      \
    	 }	\
  	 }	\
  } \
  else {  \
    for(int flatind_x = 0; flatind_x < N2; ++flatind_x) {	\
      GETX; \
    	  for(int flatind_y = 0; flatind_y < N1; ++flatind_y) {	\
    	    GETY; \
          DOCODE;                         \
          ++flatind_out;                      \
        }	\
    }	\
  } \
} while(0)



#define MACRO_DIM_TWOVECTORS(DOCODE, GETX, GETY) do {  \
  if(dimmode == 1) {  \
    MACRO_DIM_VECTOR(DOCODE, GETX, GETY); \
  } \
  else if(dimmode == 2) { \
    MACRO_DIM_ORTHOVECTOR(DOCODE, GETX, GETY);  \
  } \
  else {  \
    stop("dimmode is not a two-vector mode"); \
  } \
} while(0)






#define MACRO_DIM_BIG2VECTOR(DOCODE, GETX, GETY) do {      \
  const int N1 = INTEGER_RO(out_dim)[0];    \
  const int N2 = INTEGER_RO(out_dim)[1];    \
  const int N3 = INTEGER_RO(out_dim)[2];    \
  \
  if(vectorx) { \
    R_xlen_t flatind_y = 0;                                   \
    R_xlen_t flatind_out = 0;                                 \
    for(int iter3 = 0; iter3 < N3; ++iter3) {                 \
      for(int flatind_x = 0; flatind_x < N2; ++flatind_x) {   \
        GETX;                                                 \
        for(int iter1 = 0; iter1 <N1; ++iter1) {              \
          GETY;                                               \
          DOCODE;                                             \
          ++flatind_y;                                        \
          ++flatind_out;                                      \
        }                                                     \
      }                                                       \
    }                                                         \
  } \
  else {  \
    R_xlen_t flatind_x = 0;                                   \
    R_xlen_t flatind_out = 0;                                 \
    for(int iter3 = 0; iter3 < N3; ++iter3) {                 \
      for(int flatind_y = 0; flatind_y < N2; ++flatind_y) {   \
        GETY;                                                 \
        for(int iter1 = 0; iter1 <N1; ++iter1) {              \
          GETX;                                               \
          DOCODE;                                             \
          ++flatind_x;                                        \
          ++flatind_out;                                      \
        }                                                     \
      }                                                       \
    }                                                         \
  } \
} while(0)


#define MACRO_DIM_SANDWICH2VECTOR(DOCODE, GETX, GETY) do {      \
  \
  if(vectorx) {  \
    R_xlen_t flatind_out = 0; \
    const int *pydim = INTEGER_RO(y_dim); \
    const R_xlen_t stride_y = (double)pydim[0] * (double)pydim[1];  \
    const R_xlen_t N1 = INTEGER_RO(out_dim)[0];  \
    const int N2 = INTEGER_RO(out_dim)[1];  \
    const R_xlen_t N3 = (R_xlen_t)INTEGER_RO(out_dim)[2] * stride_y;  \
    R_xlen_t flatind_y;\
    for(R_xlen_t iter3 = 0; iter3 < N3; iter3 += stride_y) {  \
      for(int flatind_x = 0; flatind_x < N2; ++flatind_x) { \
        GETX; \
        for(R_xlen_t iter1 = 0; iter1 < N1; ++iter1) {  \
          flatind_y = iter3 + iter1; \
          GETY; \
          DOCODE; \
          ++flatind_out;  \
        } \
      } \
    } \
  } \
  else {  \
    R_xlen_t flatind_out = 0; \
    const int *pxdim = INTEGER_RO(x_dim); \
    const R_xlen_t stride_x = (double)pxdim[0] * (double)pxdim[1];  \
    const R_xlen_t N1 = INTEGER_RO(out_dim)[0];  \
    const int N2 = INTEGER_RO(out_dim)[1];  \
    const R_xlen_t N3 = INTEGER_RO(out_dim)[2] * stride_x;  \
    R_xlen_t flatind_x;\
    for(R_xlen_t iter3 = 0; iter3 < N3; iter3 += stride_x) {  \
      for(int flatind_y = 0; flatind_y < N2; ++flatind_y) { \
        GETY; \
        for(R_xlen_t iter1 = 0; iter1 < N1; ++iter1) {  \
          flatind_x = iter3 + iter1; \
          GETX; \
          DOCODE; \
          ++flatind_out;  \
        } \
      } \
    } \
  } \
} while(0)



#define MACRO_DIM_ONEVECTOR(DOCODE, GETX, GETY) do {  \
  bool vectorx = inline_dims_is_vector(x_dim);  \
  \
  if(dimmode == 3) { \
    MACRO_DIM_BIG2VECTOR(DOCODE, GETX, GETY); \
  } \
  else if(dimmode == 4) { \
    MACRO_DIM_SANDWICH2VECTOR(DOCODE, GETX, GETY); \
  } \
  else {  \
    stop("dimmode is not a one-vector mode"); \
  } \
} while(0)





#define MACRO_DIM_4(DOCODE, GETX, GETY) do {      \

  \
  	 for(int iter4 = 0; iter4 <N4; ++iter4) {	\
R_xlen_t i_x4 = stride_x[3] * iter4;	\
R_xlen_t i_y4 = stride_y[3] * iter4;	\
	 for(int iter3 = 0; iter3 <N3; ++iter3) {	\

R_xlen_t i_y3 = stride_y[2] * iter3 + i_y4;	\

R_xlen_t i_y2 = stride_y[1] * iter2 + i_y3;	\
	 for(int iter1 = 0; iter1 <N1; ++iter1) {	\
	\
	\
        R_xlen_t flatind_x = stride_x[0] * iter1 + i_x2;       \
        R_xlen_t flatind_y = stride_y[0] * iter1 + i_y2;     \
        GETX;   \
        GETY;   \
                                                                    \
        DOCODE;                                                          \
  	                                                                \
        flatind_out++;                      \
  	 }	\
	 }	\
	 }	\
	 }	\
} while(0)

#define MACRO_OVERFLOW_INT53(REF) ((REF) < int53min || (REF) > int53max)

inline double inline_int53_overflow2NA(
  double x, double int53min, double int53max
) {
  if(x < int53min) return NA_REAL;
  if(x > int53max) return NA_REAL;
  return trunc(x);
}

inline double inline_int53_overflow2Inf(
  double x, double int53min, double int53max
) {
  if(x < int53min) return R_NegInf;
  if(x > int53max) return R_PosInf;
  return trunc(x);
}


inline double inline_int53_mod(double x, double y, double int53min, double int53max) {
  if(y == 0) {
    return R_NaN;
  }
  if(MACRO_OVERFLOW_INT53(x)) {
    return R_NaN;
  }
  if(MACRO_OVERFLOW_INT53(y) && std::abs(x) <= std::abs(y)) {
    double out1 = (std::abs(x) == std::abs(y)) ? 0 :
	    ((x < 0 && y > 0) ||
	     (y < 0 && x > 0))
	     ? x+y  // differing signs
	     : x;
	   return out1;
  }
  
  double q = x / y;
  double tmp = x - floor(q) * y;
  double out = (double) (tmp - floorl(tmp / y) * y);
  return out;
}


inline double inline_int53_idiv(double x, double y, double int53min, double int53max) {
    
    double q = x / y;
    
    if (y == 0.0 || MACRO_OVERFLOW_INT53(q) || !R_FINITE(q)) {
      return q;
    }
    
    if(std::abs(q) < 1) {
      double z = (q < 0) ? -1
	    : ((x < 0 && y > 0) ||
	       (x > 0 && y < 0) // differing signs
	       ? -1 : 0);
	    return z;
    }
    
    double tmp = (double)(x - floor(q) * (double)y);
    return (double) (floor(q) + floorl(tmp/y));
  }

  if(NACHECK) {                                         \
    pout[flatind_out] = NA_REAL;                        \
  }                                                     \
  else {                                                \

} while(0)





#define TOKEN_DECLARE_double(var) const double *p##var = REAL_RO(var)
#define TOKEN_DECLARE_int(var) const int *p##var = INTEGER_RO(var)
#define TOKEN_NA_double(var) R_isnancpp(temp##var)
#define TOKEN_NA_int(var) temp##var == NA_INTEGER
#define TOKEN_STORE_double(var, OVERFLOWFUN)  const double temp##var = OVERFLOWFUN(p##var[flatind_##var], int53min, int53max)
#define TOKEN_STORE_int(var, OVERFLOWFUN)  const double temp##var = (double)p##var[flatind_##var]

#define TOKEN_TYPESWITCH(TYPE_X, TYPE_Y, DIMCODE, DOCODE, OVERFLOWFUN) do { \
  TOKEN_DECLARE_##TYPE_X(x);  \
  TOKEN_DECLARE_##TYPE_Y(y);  \
  DIMCODE(                                                  \
      MACRO_ACTION_INT_FACT(                               \
        TOKEN_NA_##TYPE_X(x) || TOKEN_NA_##TYPE_Y(y),   \
        DOCODE                                                \
      ),                                                      \
      TOKEN_STORE_##TYPE_X(x, OVERFLOWFUN),              \
      TOKEN_STORE_##TYPE_Y(y, OVERFLOWFUN)               \
    );                                                        \
} while(0)

#define MACRO_TYPESWITCH_INT_FACT(DIMCODE, DOCODE, OVERFLOWFUN) do {      \
  bool isInt_x = TYPEOF(x) == LGLSXP || TYPEOF(x) == INTSXP;    \
  bool isInt_y = TYPEOF(y) == LGLSXP || TYPEOF(y) == INTSXP;    \
  double int53max = pow(2, 53);                                 \
  double int53min = -1 * int53max;                                \
  if(TYPEOF(x) == REALSXP && TYPEOF(y) == REALSXP) {  \
      TOKEN_TYPESWITCH(double, double, DIMCODE, DOCODE, OVERFLOWFUN);  \
    }                                                   \
    else if(isInt_x && TYPEOF(y) == REALSXP) {                  \
      TOKEN_TYPESWITCH(int, double, DIMCODE, DOCODE, OVERFLOWFUN); \
    }                                                           \
    else if(TYPEOF(x) == REALSXP && isInt_y) {                  \
      TOKEN_TYPESWITCH(double, int, DIMCODE, DOCODE, OVERFLOWFUN); \
    }                                                           \
  else if(isInt_x && isInt_y) {        \
     stop("use boolmath");           \
  }                                                         \
  else {                                                    \
    stop("unsupported types given");                      \
  }                                                         \
} while(0)


#define MACRO_OP_INT_FACT(DIMCODE) do {	\
  switch(op) {	\
    case 1:	\
    {	\
      MACRO_TYPESWITCH_INT_FACT(  \
        DIMCODE,	\
        (double)rcpp_int53_gcd(tempx, tempy), \
        inline_int53_overflow2NA  \
      );    \
      break;	\
    }	\
    case 2:	\
    {	\
      MACRO_TYPESWITCH_INT_FACT(  \
          DIMCODE,	\
          inline_int53_mod(tempx, tempy, int53min, int53max), \
          inline_int53_overflow2Inf \
        );	\
      break;	\
    }	\
    case 3:	\
    {	\
      MACRO_TYPESWITCH_INT_FACT(  \
          DIMCODE,	\
         inline_int53_idiv(tempx, tempy, int53min, int53max), \
         inline_int53_overflow2Inf  \
        );	\
      break;	\
    }	\
    case 4:	\
    {	\
      MACRO_TYPESWITCH_INT_FACT( \
          DIMCODE,	\
         tempx < tempy ? tempx : tempy, \
         inline_int53_overflow2Inf  \
        );	\
      break;	\
    }	\
    case 5:	\
    {	\
      MACRO_TYPESWITCH_INT_FACT( \
          DIMCODE,	\
         tempx > tempy ? tempx : tempy, \
         inline_int53_overflow2Inf  \
        );	\
      break;	\
    }	\
    default:	\
    {	\
      stop("given operator not supported in the given context");	\
    }	\
  }	\
} while(0)




long long rcpp_int53_gcd_rec(
  long long x, long long y
) {

  long long a = std::abs(x);
  long long b = std::abs(y);
  
  if (a == 0) {
    return b;
  }
  if (b == 0) {
    return a;
  }
  if (a == b) {
    return a;
  }
  if (a > b) {
    return rcpp_int53_gcd_rec(a - b, b);
  }
  return rcpp_int53_gcd_rec(a, b - a);
}


long long rcpp_int53_gcd(
  long long x, long long y
) {
  
  if(x == 0 && y == 0) {
    return NA_REAL;
  }
  else if(x == 0) {
    return y;
  }
  else if(y == 0) {
    return x;
  }
  
  return rcpp_int53_gcd_rec(x, y);
}
